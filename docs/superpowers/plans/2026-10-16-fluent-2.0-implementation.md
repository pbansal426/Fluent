```javascript
// extension/lib/form-session.js

export class FormSession {
  constructor({ fields = [], milestones = [], metadata = {} } = {}) {
    this._fields = new Map(fields.map(f => [f.id, { ...f }]));
    this._milestones = milestones.length > 0 ? milestones : this._inferMilestones(fields);
    this._metadata = metadata;
    this._values = new Map();
  }

  static fromSchema(schema) {
    const fields = [];
    const milestones = [];

    for (const section of (schema.sections || [])) {
      const fieldIds = [];
      for (const field of (section.fields || [])) {
        fields.push({ ...field, section: section.title || section.id });
        fieldIds.push(field.id);
      }
      milestones.push({
        id: section.id,
        title: section.title || section.id,
        fieldIds,
        completed: false
      });
    }

    return new FormSession({ fields, milestones, metadata: { title: schema.title } });
  }

  static fromDOM(fields, metadata = {}) {
    return new FormSession({ fields, metadata });
  }

  static fromPDF(fields, pdfMeta = {}) {
    return new FormSession({ fields, metadata: pdfMeta });
  }

  _inferMilestones(fields) {
    const groups = new Map();
    for (const f of fields) {
      const sec = f.section || 'General Information';
      if (!groups.has(sec)) groups.set(sec, []);
      groups.get(sec).push(f.id);
    }

    let idx = 1;
    const milestones = [];
    for (const [title, fieldIds] of groups.entries()) {
      milestones.push({
        id: `milestone-${idx++}`,
        title,
        fieldIds,
        completed: false
      });
    }
    return milestones;
  }

  getFields() {
    return Array.from(this._fields.values());
  }

  getField(id) {
    return this._fields.get(id) || null;
  }

  setValue(id, value) {
    this._values.set(id, value);
    this._updateMilestoneCompletion();
  }

  getValue(id) {
    return this._values.get(id) ?? null;
  }

  getMilestones() {
    this._updateMilestoneCompletion();
    return this._milestones.map(m => ({ ...m }));
  }

  _updateMilestoneCompletion() {
    for (const m of this._milestones) {
      m.completed = m.fieldIds.every(id => {
        const f = this._fields.get(id);
        if (f && f.required) {
          const val = this._values.get(id);
          return val !== undefined && val !== null && String(val).trim() !== '';
        }
        return this._values.has(id);
      });
    }
  }

  isComplete() {
    for (const f of this._fields.values()) {
      if (f.required) {
        const val = this._values.get(f.id);
        if (val === undefined || val === null || String(val).trim() === '') {
          return false;
        }
      }
    }
    return true;
  }
}
```

```javascript
// extension/lib/metacognitive.js

const NAME_VARIANTS = [
  { pair: ['bryan', 'brian'], prompt: (w, lang) => lang === 'es' ? '¿Se escribe con Y o con I?' : 'Is that Bryan with a Y or an I?' },
  { pair: ['sara', 'sarah'], prompt: (w, lang) => lang === 'es' ? '¿Sara con o sin H al final?' : 'Is that Sarah with or without an H at the end?' },
  { pair: ['jon', 'john'], prompt: (w, lang) => lang === 'es' ? '¿Se escribe J-O-N o J-O-H-N?' : 'Is that J-O-N or J-O-H-N?' },
  { pair: ['marc', 'mark'], prompt: (w, lang) => lang === 'es' ? '¿Marc con C o con K?' : 'Is that Marc with a C or a K?' },
  { pair: ['shawn', 'sean', 'shaun'], prompt: (w, lang) => lang === 'es' ? '¿Cómo se deletrea Sean?' : 'How do you spell that Sean?' }
];

export class MetacognitiveResolver {
  constructor() {}

  assessSpelling(field, transcript, alternatives = [], lang = 'en') {
    if (!transcript || typeof transcript !== 'string') {
      return { needsConfirmation: false, value: '' };
    }

    const clean = transcript.trim();
    const lower = clean.toLowerCase();
    const isName = /name|first|last|surname|apellido|nombre/i.test(field.label || field.id || '');

    if (isName) {
      for (const v of NAME_VARIANTS) {
        if (v.pair.includes(lower) || alternatives.some(alt => v.pair.includes(alt.toLowerCase()))) {
          return {
            needsConfirmation: true,
            variantType: 'name_split',
            prompt: v.prompt(clean, lang),
            candidate: clean
          };
        }
      }
    }

    return {
      needsConfirmation: false,
      value: clean
    };
  }

  normalizeSpelledWord(input) {
    if (!input) return '';
    // Strip "as in <word>" phonetics: e.g. "B as in Boy" -> "B"
    const cleaned = input.replace(/\b(?:as in|como en)\s+[A-Za-z]+/gi, '');
    const tokens = cleaned.split(/[\s,.-]+/).filter(Boolean);
    
    // If predominantly single letters
    const singleLetters = tokens.every(t => t.length === 1);
    if (singleLetters && tokens.length > 0) {
      const word = tokens.join('');
      return word.charAt(0).toUpperCase() + word.slice(1).toLowerCase();
    }
    
    return input.trim();
  }

  normalizeEmail(transcript) {
    if (!transcript) return '';
    let email = transcript.trim().toLowerCase();
    email = email.replace(/\s+(?:at|arroba)\s+/g, '@');
    email = email.replace(/\s+(?:dot|punto)\s+/g, '.');
    email = email.replace(/\s+(?:underscore|guión bajo|guion bajo)\s+/g, '_');
    email = email.replace(/\s+(?:dash|hyphen|guión|guion)\s+/g, '-');
    email = email.replace(/\s+/g, '');
    return email;
  }
}
```

    assert.equal(current.title, 'Personal');

  it('recognizes a jump correction to an earlier field without losing current question pointer', async () => {

      { id: 'first_name', label: 'First Name', section: 'Personal' },
      { id: 'phone', label: 'Phone', section: 'Contact' }
    ];

    const agent = new Agent({
      fields,
      userLang: 'English',
      llm: {
        chat: async () => ({
          tool_calls: [
            {
              function: {
                name: 'fill_fields',
                arguments: JSON.stringify({
                  values: [{ field_id: 'first_name', value: 'Ana' }]
                })
              }
            }
          ]
        })
      }
    });

    // Fill first field
    await agent.step('Ana');
    assert.equal(agent.values['first_name'], 'Ana');

    // Currently on phone field
    assert.equal(agent.current.id, 'phone');

    // User interrupts: "Wait, my name is Maria, not Ana"
    agent.llm.chat = async () => ({
      tool_calls: [
        {
          function: {
            name: 'fill_fields',
            arguments: JSON.stringify({
              values: [{ field_id: 'first_name', value: 'Maria' }]
            })
          }
        }
      ]
    });

    const result = await agent.step('Wait, my name is Maria, not Ana');
    // Value for first_name was updated
    assert.equal(agent.values['first_name'], 'Maria');
    // Current field remains phone!
    assert.equal(agent.current.id, 'phone');
  });
});
