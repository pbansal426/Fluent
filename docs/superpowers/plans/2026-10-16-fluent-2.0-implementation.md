**Goal:** Upgrade Fluent into a universal, metacognitive form-intake engine supporting the Chrome extension, standalone web/kiosk portals, and headless/telephony schemas with phonetic spelling intelligence, milestone clustering, adaptive hesitant-speaker VAD, visual document hint cards, and an interactive bilingual verification card.

**Architecture:** Introduce `FormSession` (`extension/lib/form-session.js`) to normalize DOM, PDF, and headless JSON schemas, compute section milestones, resolve physical document hint cards, and generate air-gapped bilingual review summaries. Add `MetacognitiveResolver` (`extension/lib/metacognitive.js`) to detect phonetic spelling ambiguities in names/emails and resolve confirmations in one turn. Wire both into `Agent` (`extension/lib/agent.js`), upgrade `live.js`/`speech.js` with hesitant-speaker adaptive silence and speech ducking, and render milestone progress, document hint cards, and the bilingual verification card in `extension/panel/`.

**Tech Stack:** ES Modules (Node.js `node:test`, Chrome Manifest V3, Web Speech API, Web Audio API, HTML5/CSS3).

## Global Constraints

- Zero external npm runtime dependencies in `extension/` (native ES modules and browser APIs only).
- All 109 existing unit tests in `npm test` must continue to pass at every task boundary.
- Sensitive fields (SSN, passport, licence, card/account numbers, passwords, member IDs, EIN) are strictly air-gapped: never stored in `agent.values`, never sent to the LLM, and masked as `typed_private` in review summaries.
- No emoji icons in UI; use clean SVG or typographic indicators matching `panel.css`.
- Keep all spoken prompts at a 6th-grade reading level.

```javascript
export function getDocumentHint(field, userLang = 'English') {
  if (!field) return null;
  const text = `${field.label || ''} ${field.name || ''} ${field.htmlId || ''}`.trim();
  if (!text) return null;
  const hit = DOCUMENT_HINTS.find((h) => h.match.test(text));
  if (!hit) return null;
  const isEs = /^es|spanish|espa/i.test(String(userLang || ''));
  const copy = isEs ? hit.es : hit.en;
  return {
    id: hit.id,
    cardTitle: copy.cardTitle,
    whereToLook: copy.whereToLook,
    highlightZone: hit.highlightZone,
    sampleCode: hit.sampleCode,
  };
}

export function buildReviewItems({ fields = [], values = new Map(), filled = new Set(), translations = new Map(), typedPrivateLabel = 'typed by you, kept private' } = {}) {
  const items = [];
  for (const f of fields) {
    if (!filled.has(f.id)) continue;
    const tr = translations.get(f.id) || { label: f.label, options: f.options || [], section: f.section || '' };
    const cleanLabel = String(tr.label || f.label || '').replace(/[\s*:]+$/, '');
    const cleanOrig = String(f.label || '').replace(/[\s*:]+$/, '');
    const section = tr.section || sectionNameOf(f);
    if (f.sensitive) {
      items.push({
        id: f.id,
        label: cleanLabel,
        originalLabel: cleanOrig,
        value: '',
        shownValue: typedPrivateLabel,
        section,
        sensitive: true,
      });
      continue;
    }
    const rawVal = String(values.get(f.id) ?? '');
    let shownValue = rawVal;
    if (f.kind === 'checkbox') {
      shownValue = rawVal === 'true' ? '\u2713' : '\u2717';
    } else if (Array.isArray(f.options) && f.options.length) {
      const idx = f.options.indexOf(rawVal);
      if (idx >= 0 && Array.isArray(tr.options) && tr.options[idx]) {
        shownValue = tr.options[idx];
      }
    }
    items.push({
      id: f.id,
      label: cleanLabel,
      originalLabel: cleanOrig,
      value: rawVal,
      shownValue,
      section,
      sensitive: false,
    });
  }
  return items;
}

export class FormSession {
  constructor({ fields = [], milestones = [], metadata = {} } = {}) {
    this._fields = new Map(fields.map((f) => [f.id, { ...f }]));
    this._milestones = milestones.length > 0 ? milestones.map((m) => ({ ...m, fieldIds: [...m.fieldIds] })) : this._inferMilestones(fields);
    this._metadata = metadata;
    this._values = new Map();
    this._filled = new Set();
    this._skipped = new Set();
  }

  static fromSchema(schema = {}) {
    const fields = [];
    const milestones = [];
    for (const section of schema.sections || []) {
      const fieldIds = [];
      for (const field of section.fields || []) {
        fields.push({ ...field, section: section.title || section.id });
        fieldIds.push(field.id);
      }
      milestones.push({
        id: section.id,
        title: section.title || section.id,
        fieldIds,
      });
    }
    return new FormSession({ fields, milestones, metadata: { title: schema.title || '' } });
  }

  static fromDOM(fields = [], metadata = {}) {
    return new FormSession({ fields, metadata });
  }

  static fromPDF(fields = [], pdfMeta = {}) {
    return new FormSession({ fields, metadata: pdfMeta });
  }

  _inferMilestones(fields) {
    const groups = new Map();
    for (const f of fields) {
      const title = sectionNameOf(f);
      if (!groups.has(title)) groups.set(title, []);
      groups.get(title).push(f.id);
    }
    let idx = 1;
    const out = [];
    for (const [title, fieldIds] of groups.entries()) {
      out.push({
        id: `milestone-${idx++}`,
        title,
        fieldIds,
      });
    }
    return out;
  }

  getFields() {
    return Array.from(this._fields.values());
  }

  getField(id) {
    return this._fields.get(id) || null;
  }

  setValue(id, value) {
    const f = this._fields.get(id);
    if (!f) return;
    this._filled.add(id);
    this._skipped.delete(id);
    if (!f.sensitive) this._values.set(id, String(value));
  }

  markFilled(id, value) {
    const f = this._fields.get(id);
    if (!f) return;
    this._filled.add(id);
    this._skipped.delete(id);
    if (!f.sensitive && value !== undefined) this._values.set(id, String(value));
  }

  markSkipped(id) {
    if (!this._fields.has(id)) return;
    this._skipped.add(id);
  }

  clearField(id) {
    this._filled.delete(id);
    this._skipped.delete(id);
    this._values.delete(id);
  }

  getValue(id) {
    return this._values.get(id) ?? null;
  }

  syncState({ filled = new Set(), skipped = new Set(), values = new Map(), translations = new Map() } = {}) {
    this._filled = new Set(filled);
    this._skipped = new Set(skipped);
    this._values = new Map(values);
    this._translations = translations;
  }

  getMilestones() {
    const total = this._milestones.length;
    return this._milestones.map((m, idx) => {
      let completedCount = 0;
      for (const id of m.fieldIds) {
        if (this._filled.has(id) || this._skipped.has(id)) completedCount++;
      }
      const remainingCount = Math.max(0, m.fieldIds.length - completedCount);
      const firstField = this._fields.get(m.fieldIds[0]);
      const trSection = firstField && this._translations?.get(firstField.id)?.section;
      return {
        id: m.id,
        title: trSection || m.title,
        index: idx + 1,
        total,
        fieldIds: [...m.fieldIds],
        completedCount,
        remainingCount,
        completed: remainingCount === 0 && m.fieldIds.length > 0,
      };
    });
  }

  getCurrentMilestone(currentFieldId = null) {
    const milestones = this.getMilestones();
    if (!milestones.length) return null;
    if (currentFieldId) {
      const hit = milestones.find((m) => m.fieldIds.includes(currentFieldId));
      if (hit) return hit;
    }
    return milestones.find((m) => !m.completed) || milestones[milestones.length - 1];
  }

  isComplete() {
    for (const f of this._fields.values()) {
      if (f.required && !this._filled.has(f.id)) return false;
    }
    return this._fields.size > 0 && Array.from(this._fields.keys()).every((id) => this._filled.has(id) || this._skipped.has(id) || !this._fields.get(id).required);
  }
}
```

```javascript
export class MetacognitiveResolver {
  isNameField(field) {
    const label = String(field?.label || '');
    return NAME_FIELD.test(label) && !NOT_PERSON_NAME.test(label);
  }

  normalizeSpelledWord(input) {
    if (!input || typeof input !== 'string') return '';
    const cleaned = input.replace(/\b(?:as in|like|como en|de)\s+[A-Za-zÀ-ÿ]+/gi, ' ');
    const tokens = cleaned.split(/[\s,.-]+/).filter(Boolean);
    if (tokens.length >= 2 && tokens.every((t) => /^[A-Za-z]$/.test(t))) {
      const joined = tokens.join('');
      return joined.charAt(0).toUpperCase() + joined.slice(1).toLowerCase();
    }
    return input.trim();
  }

  normalizeEmail(transcript) {
    if (!transcript || typeof transcript !== 'string') return '';
    return transcript
      .trim()
      .toLowerCase()
      .replace(/\s+(?:at|arroba)\s+/gi, '@')
      .replace(/\s+(?:dot|punto)\s+/gi, '.')
      .replace(/\s+(?:underscore|guion bajo|guión bajo)\s+/gi, '_')
      .replace(/\s+(?:dash|hyphen|guion|guión)\s+/gi, '-')
      .replace(/\s+/g, '');
  }

  assessSpelling(field, value, { said = '', alternatives = [], lang = 'en' } = {}) {
    const cleanValue = String(value || '').trim();
    if (!cleanValue || !this.isNameField(field)) {
      return { needsConfirmation: false, value: cleanValue };
    }

    const spelled = this.normalizeSpelledWord(said);
    if (spelled !== said.trim() && spelled.length >= 2) {
      return { needsConfirmation: false, value: spelled };
    }

    const lowerVal = cleanValue.toLowerCase();
    const lowerAlts = (alternatives || []).map((a) => String(a || '').trim().toLowerCase());
    const langCode = /^es|spanish|espa/i.test(String(lang || '')) ? 'es' : 'en';

    for (const group of VARIANT_GROUPS) {
      const lowerOpts = group.options.map((o) => o.toLowerCase());
      if (lowerOpts.includes(lowerVal) || lowerAlts.some((a) => lowerOpts.includes(a))) {
        const alreadyDisambiguated = group.resolveRule(said);
        if (alreadyDisambiguated) {
          return { needsConfirmation: false, value: alreadyDisambiguated };
        }
        return {
          needsConfirmation: true,
          variantType: 'name_split',
          options: [...group.options],
          prompt: group.prompt(langCode),
          candidate: cleanValue,
          value: cleanValue,
        };
      }
    }

    return { needsConfirmation: false, value: cleanValue };
  }

  resolveConfirmation(pending, reply) {
    if (!pending || !Array.isArray(pending.options) || !reply) {
      return { resolved: false };
    }
    const text = String(reply).trim();
    const spelled = this.normalizeSpelledWord(text);
    for (const opt of pending.options) {
      if (spelled.toLowerCase() === opt.toLowerCase() || text.toLowerCase() === opt.toLowerCase()) {
        return { resolved: true, value: opt };
      }
    }

    for (const group of VARIANT_GROUPS) {
      if (group.options[0] === pending.options[0]) {
        const hit = group.resolveRule(text);
        if (hit) return { resolved: true, value: hit };
      }
    }

    for (const opt of pending.options) {
      const re = new RegExp(`\\b${opt}\\b`, 'i');
      if (re.test(text)) return { resolved: true, value: opt };
    }

    return { resolved: false };
  }
}
```

```javascript
// extension/panel/review-card.js
// Pure HTML/text formatters for the milestone progress bar, visual document hint cards,
// and the end-of-session bilingual verification sheet.

const esc = (s) =>
  String(s ?? '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');

export function renderMilestoneText(milestone, userLang = 'en') {
  if (!milestone || !milestone.total) return '';
  const isEs = /^es|spanish|espa/i.test(String(userLang || ''));
  const rem = milestone.remainingCount ?? 0;
  if (isEs) {
    const left = rem > 0 ? ` · faltan ${rem}` : '';
    return `Parte ${milestone.index} de ${milestone.total}: ${milestone.title}${left}`;
  }
  const left = rem > 0 ? ` · ${rem} left` : '';
  return `Part ${milestone.index} of ${milestone.total}: ${milestone.title}${left}`;
}

export function generateDocumentHintHtml(hint) {
  if (!hint || !hint.cardTitle) return '';
  return `<div class="doc-hint-inner" data-hint-id="${esc(hint.id)}">
    <div class="doc-hint-header">
      <span class="doc-hint-title">${esc(hint.cardTitle)}</span>
      <code class="doc-hint-sample" data-zone="${esc(hint.highlightZone || 'center')}">${esc(hint.sampleCode || '')}</code>
    </div>
    <p class="doc-hint-text">${esc(hint.whereToLook)}</p>
  </div>`;
}

export function generateBilingualReviewHtml(reviewItems = [], userLang = 'en') {
  if (!Array.isArray(reviewItems) || !reviewItems.length) return '';
  const isEs = /^es|spanish|espa/i.test(String(userLang || ''));
  const heading = isEs ? 'Resumen de verificación bilingüe' : 'Bilingual Verification Summary';
  const editLabel = isEs ? 'Cambiar' : 'Edit';

  const rows = reviewItems
    .map(
      (item) => `<div class="review-row${item.sensitive ? ' is-private' : ''}">
      <div class="review-labels">
        <span class="review-user-label">${esc(item.label)}</span>
        <span class="review-orig-label">${esc(item.originalLabel)}</span>
      </div>
      <div class="review-value-wrap">
        <span class="review-val">${esc(item.shownValue || item.value)}</span>
        <button type="button" class="small review-edit" data-edit-field="${esc(item.id)}">${esc(editLabel)}</button>
      </div>
    </div>`
    )
    .join('');

  return `<div class="review-sheet">
    <div class="review-title">${esc(heading)}</div>
    <div class="review-list">${rows}</div>
  </div>`;
}
```
